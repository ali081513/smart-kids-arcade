            function checkWin(b) {

                const lines = [[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];

                for(let l of lines) if(b[l[0]] && b[l[0]]===b[l[1]] && b[l[0]]===b[l[2]]) return b[l[0]];

                if(!b.includes(null)) return 'draw'; return null;

            }

            function getBestMove() {

                const lines = [[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];

                // Check if CPU can win

                for(let l of lines) {

                    let [a,b,c] = l;

                    if(board[a]==='❌' && board[b]==='❌' && !board[c]) return c;

                    if(board[a]==='❌' && board[c]==='❌' && !board[b]) return b;

                    if(board[b]==='❌' && board[c]==='❌' && !board[a]) return a;

                }

                // Block Player

                for(let l of lines) {

                    let [a,b,c] = l;

                    if(board[a]==='⭕' && board[b]==='⭕' && !board[c]) return c;

                    if(board[a]==='⭕' && board[c]==='⭕' && !board[b]) return b;

                    if(board[b]==='⭕' && board[c]==='⭕' && !board[a]) return a;

                }

                // Random center or edges

                let e = board.map((c,idx)=>c===null?idx:null).filter(c=>c!==null);

                if(e.includes(4)) return 4;

                return e[randInt(0, e.length-1)];

            }

            function play(i) {

                board[i] = '⭕'; render(); let w = checkWin(board); if(w) return end(w);

                let e = board.map((c,idx)=>c===null?idx:null).filter(c=>c!==null);

                if(e.length > 0) {

                    board[getBestMove()] = '❌';

                    setTimeout(() => { render(); w = checkWin(board); if(w) end(w); }, 300);

                }

            }

            function end(r) {

                gameOver = true;

                    if(r==='draw') showModal('🤝', loc('draw'), '', initTTT);

                }, 100);

            }

            render();

        }

                <canvas id="snake-canvas" width="300" height="300" class="bg-green-100 border-4 border-green-500 rounded-lg shadow-inner touch-none"></canvas>

            `;

            

            const cvs = document.getElementById('snake-canvas');

            const ctx = cvs.getContext('2d');

            let gridSize = 15;

            let tileCount = cvs.width / gridSize;

            let snake = [{x: 10, y: 10}], food = {x: 5, y: 5};

            let dx = 0, dy = 0, score = 0;

            let gameLoop;

            

            function draw() {

                let head = {x: snake[0].x + dx, y: snake[0].y + dy};

                

                // Wrap around edges

                if(head.x < 0) head.x = tileCount-1;

                if(head.x >= tileCount) head.x = 0;

                if(head.y < 0) head.y = tileCount-1;

                if(head.y >= tileCount) head.y = 0;


                // Check collision with self

                if(dx !== 0 || dy !== 0) {

                    for(let i=0; i<snake.length; i++) {

                        if(head.x === snake[i].x && head.y === snake[i].y) {

                            clearInterval(gameLoop);

                            return showModal('💥', loc('done'), `${loc('scoreText')}: ${score}`, initSnake);

                        }

                    }

                }

                

                snake.unshift(head);

                

                if(head.x === food.x && head.y === food.y) {

                    score++; document.getElementById('snk-score').textContent = score;

                    food = {x: randInt(0, tileCount-1), y: randInt(0, tileCount-1)};

                } else {

                    if(dx !== 0 || dy !== 0) snake.pop();

                }

                

                ctx.fillStyle = '#dcfce7'; // green-100

                ctx.fillRect(0,0, cvs.width, cvs.height);

                

                ctx.fillStyle = '#ef4444'; // red-500 food

                ctx.fillRect(food.x*gridSize, food.y*gridSize, gridSize-2, gridSize-2);

                

                ctx.fillStyle = '#22c55e'; // green-500 snake

                snake.forEach(p => {

                    ctx.fillRect(p.x*gridSize, p.y*gridSize, gridSize-2, gridSize-2);

                });

            }

            

            gameLoop = registerInterval(draw, 120);

            

            window.onkeydown = (e) => {

                if(e.key==='ArrowUp' && dy===0) { dx=0; dy=-1; }

                if(e.key==='ArrowDown' && dy===0) { dx=0; dy=1; }

                if(e.key==='ArrowLeft' && dx===0) { dx=-1; dy=0; }

                if(e.key==='ArrowRight' && dx===0) { dx=1; dy=0; }

            };

            

            // Touch Swipe Logic

            let touchStartX = 0, touchStartY = 0;

            cvs.addEventListener('touchstart', e => {

                e.preventDefault();

                touchStartX = e.changedTouches[0].screenX;

                touchStartY = e.changedTouches[0].screenY;

            }, {passive: false});

            cvs.addEventListener('touchend', e => {

                e.preventDefault();

                let endX = e.changedTouches[0].screenX, endY = e.changedTouches[0].screenY;

                let diffX = endX - touchStartX, diffY = endY - touchStartY;

                if(Math.abs(diffX) > Math.abs(diffY)) {

                    if(diffX > 30 && dx===0) { dx=1; dy=0; }

                    else if(diffX < -30 && dx===0) { dx=-1; dy=0; }

                } else {

                    if(diffY > 30 && dy===0) { dx=0; dy=1; }

                    else if(diffY < -30 && dy===0) { dx=0; dy=-1; }

                }

            }, {passive: false});

        }
